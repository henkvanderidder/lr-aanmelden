4 controleer Installatie supervisor
sudo apt install supervisor

5 controleer voeg toe aan /etc/supervisord/conf.d/queueworker.conf
[program:queue-worker]
process_name=%(program_name)s_%(process_num)02d
command=php /var/www/snipe-it/artisan queue:work --sleep=3 --tries=3 --max-time=3600
autostart=true
autorestart=true
stopasgroup=true
killasgroup=true
numprocs=1
redirect_stderr=true
stdout_logfile=/var/www/snipe-it/storage/logs/queueworker.log
stopwaitsecs=3600

6 controleer Supervisor configureren
 supervisorctl reread
 supervisorctl update
 supervisorctl restart all

7 Controleer inhoud in crontab
* * * * * root  (cd /var/www/snipe-it && php artisan schedule:run) >>/var/www/snipe-it/storage/logs/cron.log 2>&1
